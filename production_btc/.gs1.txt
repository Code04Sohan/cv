/**
 * Verifies the incoming token matches the single active device session.
 */
function validateSession(incomingToken) {
  if (!incomingToken) return false;
  const activeToken = getProps().getProperty("SESSION_TOKEN");
  return (incomingToken === activeToken);
}

/**
 * Handles user login and enforces the Single-Session policy.
 */
function handleLogin(payload) {
  const props = getProps();
  let storedPass = props.getProperty("ADMIN_PASS");
  
  if (!storedPass) {
    // Graceful initialization fallback to "admin123" if script properties aren't set yet
    storedPass = "admin123";
    props.setProperty("ADMIN_PASS", storedPass);
  }

  if (payload.password === storedPass) {
    const newToken = Utilities.getUuid();
    props.setProperty("SESSION_TOKEN", newToken); // Invalidates old sessions
    return createResponse({ status: "success", token: newToken });
  }
  
  return createResponse({ status: "error", message: "Invalid Security Key." });
}

/**
 * ==========================================
 * 🚦 MODULE 3: ROUTING ENGINE (GATEKEEPER)
 * ==========================================
 */
function doPost(e) {
  try {
    let payload;
    if (e.postData && e.postData.contents) {
      try {
        payload = JSON.parse(e.postData.contents);
      } catch (err) {
        payload = e.parameter;
      }
    } else {
      payload = e.parameter;
    }
    
    if (!payload || !payload.action) {
        return createResponse({ status: "error", message: "Empty or invalid payload." });
    }
    
    const action = payload.action;

    // A. PUBLIC ROUTES (No auth required)
    if (action === "LOGIN") return handleLogin(payload);
    if (action === "REQUEST_RESET") return requestReset(payload);
    if (action === "COMPLETE_RESET") return verifyAndReset(payload);

    // B. THE GATEKEEPER - Enforce token checks on secure endpoints
    if (!validateSession(payload.token)) {
      return createResponse({ status: "error", message: "Unauthorized or Expired Session." });
    }

    // C. SECURE ROUTES
    switch (action) {
      case "VALIDATE_SESSION":
        return createResponse({ status: "success", message: "Session valid." });
      
      // Candidate Ingestion Engine integration
      case "ADD_CANDIDATE":
        return handleAddCandidate(payload.candidate);
        
      case "FETCH_DIRECTORY":
        return handleFetchDirectory(payload);
        
      case "EDIT_RECORD":
        return handleEditRecord(payload.record);

      case "GET_IMAGE_BASE64":
        return handleGetImageBase64(payload);

      case "DELETE_RECORD":
        return handleDeleteRecord(payload);

      // ====================================================
      // 💳 SHARDED PAYMENT LEDGER MODULE ROUTES (Logs_YYYY)
      // ====================================================
      case "GET_FEE_CONFIG":
        return handleGetFeeConfig();

      case "SET_FEE_CONFIG":
        return handleSetFeeConfig(payload.monthlyFee, payload.admissionFee);

      case "BULK_LOG_PAYMENTS":
        return handleBulkLogPayments(payload.payloadArray);

      case "LAZY_FETCH_STUDENT_LOGS":
        return handleLazyFetchStudentLogs(payload.studentId, payload.targetYear);

      case "FETCH_STATEMENT_RANGE":
        return handleFetchStatementRange(payload.startDate, payload.endDate);

      case "SEND_RUNTIME_EMAIL_ATTACHMENT":
        return handleSendRuntimeEmailAttachment(payload);

      case "FETCH_MONTHLY_LEDGER":
        return handleFetchMonthlyLedger(payload);

      default:
        return createResponse({ status: "error", message: "Unknown action." });
    }
  } catch (error) {
    return createResponse({ status: "error", message: "Engine Error: " + error.toString() });
  }
}

/**
 * handleFetchMonthlyLedger
 * -------------------------------------------------------------------
 * Opens the target annual shard (Logs_YYYY) and isolates transaction records
 * matching the targeted numeric month parameter using fast string matching.
 * 
 * @param {Object} requestPayload Contains targetYear, targetMonth (string format, e.g. "07")
 */
function handleFetchMonthlyLedger(requestPayload) {
  try {
    var targetYear = requestPayload.year || String(new Date().getFullYear());
    var targetMonth = requestPayload.month; // Expected zero-padded string format, e.g., "01" through "12"

    // ─────────────────────────────────────────────────────────────────────
    // 🚀 ADVANCED SHEETS API v4 — High-performance read path
    // Resolve the correct shard tab name with smart fallback convention.
    // The Advanced API fetches the raw 2D array directly from the Sheets
    // REST endpoint, bypassing the SpreadsheetApp object model overhead.
    // ─────────────────────────────────────────────────────────────────────

    // 2. Smart Sheet Tab Fallback — determine which named range to target
    var candidateSheetNames = ["Logs_" + targetYear, "Logs", "Payment_Logs"];
    var resolvedSheetName = null;

    for (var s = 0; s < candidateSheetNames.length; s++) {
      var candidateName = candidateSheetNames[s];
      try {
        // A lightweight probe: ask for a single cell; if the sheet is absent
        // the API throws rather than returns an empty body.
        var probe = Sheets.Spreadsheets.Values.get(
          CONFIG.SPREADSHEET_ID,
          "'" + candidateName + "'!A1"
        );
        // If we reach here, the sheet exists
        resolvedSheetName = candidateName;
        break;
      } catch (probeErr) {
        // Sheet does not exist — try next candidate
      }
    }

    // Safety fallback if no tabs exist
    if (!resolvedSheetName) {
      return createResponse({ status: "success", data: [], message: "Log sheet for year " + targetYear + " not found." });
    }

    // Full shard read via Advanced Sheets API
    var response = Sheets.Spreadsheets.Values.get(
      CONFIG.SPREADSHEET_ID,
      "'" + resolvedSheetName + "'"
    );

    var rawData = (response && response.values) ? response.values : [];

    if (rawData.length <= 1) {
      return createResponse({ status: "success", data: [] });
    }

    var headers = rawData[0];

    // 1. Smart Column Header Detection
    var getColIndex = function(headerArray, possibleNames) {
      for (var i = 0; i < possibleNames.length; i++) {
        var idx = headerArray.indexOf(possibleNames[i]);
        if (idx !== -1) return idx;
      }
      return -1;
    };

    var txnIdIdx       = headers.indexOf("TXN_ID");
    var studentNameIdx = headers.indexOf("STUDENT_NAME");
    var feePeriodIdx   = headers.indexOf("FEE_PERIOD");
    var amountIdx      = headers.indexOf("AMOUNT_COLLECTED");

    var studentIdIdx = getColIndex(headers, ["STUDENT_ID", "ID", "REGISTRATION_NO"]);
    var dateIdx      = getColIndex(headers, ["TRANSACTION_DATE", "DATE", "TIMESTAMP", "PAYMENT_DATE", "DATE_OF_PAYMENT"]);

    var filteredMonthlyLogs = [];

    // Note: Advanced API returns all values as strings; Date objects are not returned.
    // We perform universal string-based month extraction on the raw cell value.

    // Loop through database records backwards to maintain recent-first order natively
    for (var i = rawData.length - 1; i >= 1; i--) {
      var row = rawData[i];
      var rawDateValue = (dateIdx !== -1 && dateIdx < row.length) ? row[dateIdx] : "";

      if (!rawDateValue) continue;

      // 3. Universal Date/Month Extractor (string-safe for Advanced API output)
      var parsedMonth = "";
      var dateString  = String(rawDateValue).trim();

      // Try ISO format YYYY-MM-DD (Advanced API canonical output)
      var isoMatch = dateString.match(/^(\d{4})-(\d{2})-(\d{2})/);
      if (isoMatch) {
        parsedMonth = isoMatch[2]; // Already zero-padded
      } else {
        // Covers DD/MM/YYYY or DD-MM-YYYY legacy formats
        var parts = dateString.split(/[\/\-]/);
        if (parts.length >= 3) {
          var extracted = String(parts[1]).trim();
          parsedMonth = extracted.length === 1 ? "0" + extracted : extracted;
        }
      }

      // 4. Absolute Wildcard Check
      var isWildcard = !targetMonth || targetMonth === "ALL" || targetMonth === "All Months" || targetMonth === "";

      // SERVER-SIDE PRE-FILTER MATCH: Skip rows that do not match the selected target month
      if (!isWildcard && parsedMonth !== targetMonth) {
        continue;
      }

      filteredMonthlyLogs.push({
        TXN_ID:           txnIdIdx       !== -1 && txnIdIdx       < row.length ? String(row[txnIdIdx])       : "",
        STUDENT_NAME:     studentNameIdx !== -1 && studentNameIdx < row.length ? String(row[studentNameIdx]) : "",
        STUDENT_ID:       studentIdIdx   !== -1 && studentIdIdx   < row.length ? String(row[studentIdIdx])   : "",
        FEE_PERIOD:       feePeriodIdx   !== -1 && feePeriodIdx   < row.length ? String(row[feePeriodIdx])   : "",
        AMOUNT_COLLECTED: amountIdx      !== -1 && amountIdx      < row.length ? Number(row[amountIdx])      : 0,
        TRANSACTION_DATE: dateString
      });
    }

    return createResponse({
      status: "success",
      data: filteredMonthlyLogs
    });

  } catch (error) {
    return createResponse({
      status: "error",
      message: "Server Monthly Ledger Query Pipeline Failure: " + error.toString()
    });
  }
}


/**
 * ==========================================
 * 🗑️ RECORD SOFT-DELETION ENGINE (IS_DELETED Flag Writer)
 * ==========================================
 * Architectural Contract:
 *   ZERO STRUCTURAL ROW SHIFTS. This function NEVER calls sheet.deleteRow().
 *   Instead it locates the target student profile row and writes boolean TRUE
 *   into the designated IS_DELETED column cell, archiving the record in-place.
 *
 * Execution Pipeline:
 *   1. Resolve the IS_DELETED column index dynamically from the live header row.
 *      This makes the engine immune to future column reordering.
 *   2. Scan Column A (STUDENT_ID) for an exact match via TextFinder.
 *   3. Guard: reject if student is already soft-deleted (idempotent safety).
 *   4. Guard: reject if IS_DELETED column does not exist in the sheet schema
 *      (prevents silent data writes to the wrong cell).
 *   5. Write boolean TRUE to the IS_DELETED cell via a single targeted setValues()
 *      call — minimum surface area, maximum write speed.
 *
 * Downstream Effects:
 *   • handleFetchDirectory() server-side pre-filter will silently discard
 *     this row on all subsequent directory ingestion calls.
 *   • Historical Logs_YYYY payment ledger shards are completely unaffected.
 *     They cross-reference by STUDENT_ID only — no row index dependency.
 *   • handleEditRecord() continues to work against this row if an admin
 *     needs to inspect or restore the profile in a future Phase.
 */
function handleDeleteRecord(payload) {
  try {
    var spreadsheet = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID);
    var sheet = spreadsheet.getSheetByName(CONFIG.MASTER_SHEET_NAME);

    if (!sheet) {
      return createResponse({ success: false, status: "error", message: "Error: Database sheet not found." });
    }

    var studentId = String(payload.STUDENT_ID);

    // ── STEP 1: Pull full data range into RAM in a single API call ─────────
    var rawData = sheet.getDataRange().getValues();

    if (rawData.length <= 1) {
      return createResponse({ success: false, status: "error", message: "Error: Directory is empty." });
    }

    var headers = rawData[0];

    // ── STEP 2: Resolve IS_DELETED column index from the live header row ───
    // Dynamic resolution ensures correctness regardless of column ordering.
    var deletedColIdx = headers.indexOf("IS_DELETED");

    if (deletedColIdx === -1) {
      // Schema guard: IS_DELETED column has not yet been added to the sheet.
      // Instruct the administrator to append the column before soft-deletes
      // can be committed — prevents writing TRUE into the wrong cell.
      return createResponse({
        success: false,
        status: "error",
        message: "Schema Error: IS_DELETED column not found in sheet headers. " +
                 "Please add an IS_DELETED column as the last column of the '" +
                 CONFIG.MASTER_SHEET_NAME + "' sheet before performing deletions."
      });
    }

    // ── STEP 3: Locate target row by STUDENT_ID match ──────────────────────
    var targetRowIndex = -1; // 0-based within rawData (includes header at [0])

    for (var i = 1; i < rawData.length; i++) {
      if (String(rawData[i][0]).trim() === studentId.trim()) {
        targetRowIndex = i;
        break;
      }
    }

    // Validation Gate: student not found
    if (targetRowIndex === -1) {
      return createResponse({ success: false, status: "error", message: "Error: Student ID '" + studentId + "' not found in database." });
    }

    // ── STEP 4: Idempotency Guard ───────────────────────────────────────────
    // If the row is already soft-deleted, reject silently to prevent
    // duplicate archival signals and unnecessary write quota consumption.
    var existingFlag = rawData[targetRowIndex][deletedColIdx];
    if (existingFlag === true || String(existingFlag).toUpperCase() === "TRUE") {
      return createResponse({
        success: false,
        status: "error",
        message: "Conflict: Student ID '" + studentId + "' has already been archived (IS_DELETED = TRUE)."
      });
    }

    // ── STEP 5: Write IS_DELETED = TRUE (Soft-Delete Commit) ───────────────
    // Convert both indexes to 1-based spreadsheet coordinates.
    //   targetRowIndex is 0-based in rawData (header = row 0),
    //   so the sheet row number = targetRowIndex + 1.
    //   deletedColIdx is 0-based, so the sheet column number = deletedColIdx + 1.
    var sheetRowNumber = targetRowIndex + 1;
    var sheetColNumber = deletedColIdx + 1;

    // Targeted single-cell write — minimum API surface, zero row shifting.
    sheet.getRange(sheetRowNumber, sheetColNumber).setValue(true);

    return createResponse({
      success: true,
      status: "success",
      message: "Student ID '" + studentId + "' has been soft-deleted and archived successfully. " +
               "The physical row is preserved. The profile will no longer appear in the active directory."
    });

  } catch (error) {
    return createResponse({
      success: false,
      status: "error",
      message: "Soft-Delete Engine Exception: " + error.toString()
    });
  }
}

const OTP_EXPIRY_MS = 5 * 60 * 1000; // 5 minutes

function requestReset(payload) {
  const props = getProps();
  const otp = Math.floor(100000 + Math.random() * 900000).toString();
  
  props.setProperty("RESET_OTP", otp);
  props.setProperty("OTP_TIMESTAMP", new Date().getTime().toString());
  
  try {
    MailApp.sendEmail({
      to: CONFIG.ADMIN_EMAIL,
      subject: "Babla Yoga Training Center ERP Portal - Password Reset Required",
      body: `Your secure identity override verification token is: ${otp}\n\nThis token expires in 5 minutes.`
    });
    return createResponse({ status: "success", message: "OTP transmitted." });
  } catch (e) {
    return createResponse({ status: "error", message: "MailApp quota exceeded or unauthorized: " + e.toString() });
  }
}

function verifyAndReset(payload) {
  const props = getProps();
  const savedOtp = props.getProperty("RESET_OTP");
  const savedTimestamp = parseInt(props.getProperty("OTP_TIMESTAMP") || "0", 10);
  
  if (savedOtp && payload.otp === savedOtp && (new Date().getTime() - savedTimestamp) < OTP_EXPIRY_MS) {
    props.setProperty("ADMIN_PASS", payload.newPassword);
    props.deleteProperty("RESET_OTP");
    props.deleteProperty("OTP_TIMESTAMP");
    props.deleteProperty("SESSION_TOKEN"); // Terminate all other sessions
    return createResponse({ status: "success", message: "Security Key updated." });
  }
  return createResponse({ status: "error", message: "Invalid or expired OTP." });
}

/**
 * Converts a raw base64 string into a drive file, returning its public sharing URL.
 * Handles folder search and creation with multi-layered fallbacks.
 */
function processBase64File(fileName, mimeType, base64Data) {
  if (!base64Data) return "";
  
  const cleanData = base64Data.split(',').pop(); 
  const decoded = Utilities.base64Decode(cleanData);
  const blob = Utilities.newBlob(decoded, mimeType, fileName);
  
  // Auto-Healing Storage: get or create folder by name directly
  const parentFolder = getOrCreateDriveFolder("Babla Yoga Candidate Documents");
  
  const file = parentFolder.createFile(blob);
  
  // Set sharing properties viewable by anyone with the link
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  
  return `https://drive.usercontent.google.com/download?id=${file.getId()}&export=view`;
}

function getOrCreateDriveFolder(folderName) {
  var folders = DriveApp.getFoldersByName(folderName);
  if (folders.hasNext()) {
    return folders.next();
  }
  return DriveApp.createFolder(folderName);
}

/**
 * handleFetchDirectory — Directory Ingestion Pipeline (Soft-Delete Aware)
 * ─────────────────────────────────────────────────────────────────────────
 * Reads the master student registry from the spreadsheet, resolves the
 * IS_DELETED column index dynamically at runtime, and enforces a
 * SERVER-SIDE PRE-FILTER that silently discards every soft-deleted row
 * before the JSON response payload is compiled.
 *
 * Why server-side?  Keeping deleted profiles off the wire ensures:
 *   • Minimal network payload to the browser cache.
 *   • Autocomplete, search, and all client modules stay clean without
 *     requiring any frontend rewrites or filter decorators.
 *   • Historical financial ledger tabs (Logs_YYYY) are completely
 *     unaffected — they carry their own STUDENT_ID cross-reference
 *     and will continue to process normally.
 *
 * Pre-Filter Gate Logic:
 *   row[deletedIdx] === true          → boolean TRUE  (Sheets checkbox)
 *   String(row[deletedIdx]).toUpperCase() === "TRUE" → string "TRUE"
 *   Either condition → row skipped; never enters activeCandidates[].
 *
 * Backwards Compatibility:
 *   If IS_DELETED column is absent (deletedIdx === -1), the guard is a
 *   no-op and all rows are returned — matching the previous behaviour
 *   so that existing deployments without the column are unharmed.
 */
function handleFetchDirectory(payload) {
  try {
    // ─────────────────────────────────────────────────────────────────────
    // 🚀 ADVANCED SHEETS API v4 — Global Candidate Directory Read
    // Replaces SpreadsheetApp.getDataRange().getValues() with a direct
    // REST call to Sheets.Spreadsheets.Values.get. This bypasses the
    // heavyweight SpreadsheetApp object model and delivers raw 2D array
    // data at native REST throughput — critical for large directories.
    //
    // UI Data Contract: The response shape is preserved exactly.
    //   • Row[0]  → treated as headers.
    //   • Row[1…] → mapped into { header: value } objects.
    //   • IS_DELETED column is resolved dynamically and stripped from
    //     the payload; soft-deleted rows are silently discarded server-side.
    // ─────────────────────────────────────────────────────────────────────
    const response = Sheets.Spreadsheets.Values.get(
      CONFIG.SPREADSHEET_ID,
      "'" + CONFIG.MASTER_SHEET_NAME + "'"
    );

    const rawData = (response && response.values) ? response.values : [];

    if (rawData.length <= 1) {
      return createResponse({ status: "success", data: [] });
    }

    const headers = rawData[0];

    // ── Resolve the IS_DELETED column index dynamically ──────────────────
    // Using indexOf() at runtime means the filter is immune to future column
    // reordering as long as the header string remains "IS_DELETED".
    const deletedIdx = headers.indexOf("IS_DELETED");

    const activeCandidates = [];

    for (let i = 1; i < rawData.length; i++) {
      const row = rawData[i];

      // ── SERVER-SIDE PRE-FILTER ────────────────────────────────────────
      // Skip this row entirely if the IS_DELETED cell is flagged true.
      // Advanced API returns booleans as the string "TRUE" / "FALSE";
      // we normalise both native boolean (legacy path) and string forms.
      if (deletedIdx !== -1 && deletedIdx < row.length) {
        const deletedFlag = row[deletedIdx];
        if (deletedFlag === true || String(deletedFlag).toUpperCase() === "TRUE") {
          continue; // Row is soft-deleted — discard silently, never stream
        }
      }

      // ── DYNAMIC HEADER-MAPPED ROW PACK ───────────────────────────────
      // Build the record object against the live runtime headers so that
      // any non-IS_DELETED columns added to the sheet in future releases
      // are automatically included without requiring a backend redeploy.
      // NOTE: Advanced API may return shorter rows for trailing empty cells;
      // we default missing cells to an empty string for contract stability.
      const record = {};
      for (let j = 0; j < headers.length; j++) {
        const header = headers[j];
        if (header === "IS_DELETED") continue; // Strip flag — internal metadata only
        record[header] = (j < row.length) ? row[j] : "";
      }

      activeCandidates.push(record);
    }

    return createResponse({ status: "success", data: activeCandidates });

  } catch (error) {
    return createResponse({
      status: "error",
      message: "Directory Ingestion Pipeline Failure: " + error.toString()
    });
  }
}

/**
 * Edits a specific record based on STUDENT_ID
 */
function handleEditRecord(record) {
  const spreadsheet = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID);
  const sheet = spreadsheet.getSheetByName(CONFIG.MASTER_SHEET_NAME);
  
  if (!sheet) {
    return createResponse({ status: "error", message: "Database not found." });
  }
  
  const dataRange = sheet.getDataRange();
  const allValues = dataRange.getValues();
  const headers = allValues[0];
  
  const studentIdIndex = headers.indexOf("STUDENT_ID");
  if (studentIdIndex === -1) {
    return createResponse({ status: "error", message: "Invalid schema mapping." });
  }
  
  let rowIndex = -1;
  for (let i = 1; i < allValues.length; i++) {
    if (String(allValues[i][studentIdIndex]) === String(record.STUDENT_ID)) {
      rowIndex = i + 1; // 1-indexed for spreadsheet, plus 1 for header
      break;
    }
  }
  
  if (rowIndex === -1) {
    return createResponse({ status: "error", message: "Student record not found." });
  }
  
  // Convert new base64 files if any
  if (record.STUDENT_PHOTO_URL && record.STUDENT_PHOTO_URL.startsWith("data:")) {
    record.STUDENT_PHOTO_URL = uploadCandidateBase64(record.STUDENT_PHOTO_URL, "STUDENT_" + record.STUDENT_NAME.replace(/\s+/g, "_") + "_" + record.STUDENT_ID);
  }
  
  if (record.STUDENT_SIGNATURE_URL && record.STUDENT_SIGNATURE_URL.startsWith("data:")) {
    record.STUDENT_SIGNATURE_URL = uploadCandidateBase64(record.STUDENT_SIGNATURE_URL, "SIGNATURE_" + record.STUDENT_NAME.replace(/\s+/g, "_") + "_" + record.STUDENT_ID);
  }
  
  const rowDataToSave = [];
  for (let j = 0; j < headers.length; j++) {
    const col = headers[j];
    // Keep old value if not provided in update, except when explicitly overwritten
    if (record[col] !== undefined) {
      rowDataToSave.push(record[col]);
    } else {
      rowDataToSave.push(allValues[rowIndex - 2][j]); 
    }
  }
  
  // Single bulk write back to the row
  sheet.getRange(rowIndex, 1, 1, headers.length).setValues([rowDataToSave]);
  
  return createResponse({ status: "success", message: "Record updated successfully." });
}

/**
 * ==========================================
 * 🖼️ MODULE 6B: SERVER-SIDE CORS IMAGE PROXY
 * ==========================================
 * Bypasses browser CORS entirely by fetching Google Drive file blobs
 * on the Apps Script server and returning them as Base64 data URIs.
 */

/**
 * Extracts a Google Drive File ID from any common Drive URL pattern,
 * fetches the file blob server-side, and returns a clean Base64 data URI.
 * Supports: /file/d/ID, open?id=ID, id=ID&export, uc?id=ID, download?id=ID
 */
function handleGetImageBase64(payload) {
  const url = payload.url;

  if (!url || typeof url !== 'string' || url.trim() === '') {
    return createResponse({ status: "error", message: "No URL provided." });
  }

  // If it's already a data URI, pass it straight back
  if (url.startsWith('data:')) {
    return createResponse({ status: "success", base64: url });
  }

  // Extract Google Drive File ID using multi-pattern regex matching
  var fileId = null;
  var patterns = [
    /\/d\/([a-zA-Z0-9_-]+)/,         // /file/d/FILE_ID/
    /[?&]id=([a-zA-Z0-9_-]+)/,        // ?id=FILE_ID or &id=FILE_ID
    /\/open\?id=([a-zA-Z0-9_-]+)/,    // /open?id=FILE_ID
    /uc\?.*id=([a-zA-Z0-9_-]+)/       // uc?export=download&id=FILE_ID
  ];

  for (var i = 0; i < patterns.length; i++) {
    var match = url.match(patterns[i]);
    if (match && match[1]) {
      fileId = match[1];
      break;
    }
  }

  if (!fileId) {
    return createResponse({ status: "error", message: "Could not extract Drive File ID from URL: " + url });
  }

  try {
    var file = DriveApp.getFileById(fileId);
    var blob = file.getBlob();
    var mimeType = blob.getContentType() || file.getMimeType();
    var bytes = blob.getBytes();
    var base64String = Utilities.base64Encode(bytes);
    var dataUri = "data:" + mimeType + ";base64," + base64String;

    return createResponse({ status: "success", base64: dataUri });
  } catch (err) {
    console.error("CORS Proxy fetch failed for fileId: " + fileId, err);
    return createResponse({ status: "error", message: "Drive file access error: " + err.toString() });
  }
}

/**
 * Parses files safely into Drive blobs and appends the fully populated candidate MasterRecord.
 */
function handleAddCandidate(candidate) {
  const spreadsheet = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID);
  let sheet = spreadsheet.getSheetByName(CONFIG.MASTER_SHEET_NAME);
  
  if (!sheet) {
    sheet = spreadsheet.insertSheet(CONFIG.MASTER_SHEET_NAME);
  }
  
  // Auto-Healing Database: Ensure headers exist if sheet is completely blank
  if (sheet.getLastRow() === 0 && sheet.getLastColumn() === 0) {
    const headers = SHEET_TEMPLATES.MasterRecord;
    sheet.appendRow(headers);
    sheet.setFrozenRows(1);
  }
  
  // ⚡ OPTIMIZED MEMORY READS: Pull database rows into RAM in a single partitioning call
  const dataRange = sheet.getDataRange();
  const allValues = dataRange.getValues();
  const headers = allValues[0];
  
  // Strict Multi-Column Duplicate Check
  if (allValues.length > 1) {
    const excludedFields = ['STUDENT_ID', 'TIMESTAMP', 'STUDENT_PHOTO_URL', 'STUDENT_SIGNATURE_URL'];
    const fieldsToCheck = headers.filter(h => !excludedFields.includes(h));
    
    const isDuplicate = allValues.slice(1).some(function(row) {
      return fieldsToCheck.every(function(field) {
        const fieldIndex = headers.indexOf(field);
        const rowVal = String(row[fieldIndex] || "").trim();
        const candidateVal = String(candidate[field] || "").trim();
        return rowVal === candidateVal;
      });
    });

    if (isDuplicate) {
      return createResponse({
        status: "error",
        message: "Duplicate student detected: An identical application with the same demographic and parental details already exists."
      });
    }
  }

  // 📂 FILE STREAMING: Convert base64 fields dynamically using our resilient processBase64File
  candidate.STUDENT_PHOTO_URL = uploadCandidateBase64(candidate.STUDENT_PHOTO_URL, "STUDENT_" + candidate.STUDENT_NAME.replace(/\s+/g, "_") + "_" + candidate.STUDENT_ID);
  candidate.STUDENT_SIGNATURE_URL = uploadCandidateBase64(candidate.STUDENT_SIGNATURE_URL, "SIGNATURE_" + candidate.STUDENT_NAME.replace(/\s+/g, "_") + "_" + candidate.STUDENT_ID);

  // Map candidate parameters to schema header template exactly
  const rowDataToSave = [];
  for (let i = 0; i < headers.length; i++) {
    const col = headers[i];
    if (col === "TIMESTAMP") {
      rowDataToSave.push(candidate.TIMESTAMP ? new Date(candidate.TIMESTAMP) : new Date());
    } else {
      rowDataToSave.push(candidate[col] !== undefined ? candidate[col] : "");
    }
  }
  
  // ⚡ HIGH-SPEED APPENDS: Single isolated write to Sheet
  sheet.appendRow(rowDataToSave);
  
  return createResponse({
    status: "success",
    message: `Candidate ${candidate.STUDENT_NAME} enrolled successfully!`,
    studentId: candidate.STUDENT_ID
  });
}

/**
 * Extracts mimetype and base64 fields, sending it to processBase64File
 */
function uploadCandidateBase64(base64Str, baseFilename) {
  if (!base64Str || !base64Str.startsWith("data:")) {
    return base64Str || ""; // Returns as-is if empty or already uploaded
  }
  
  try {
    const parts = base64Str.split(",");
    const mimeType = parts[0].match(/data:([^;]+);/)[1];
    const extension = mimeType.split("/")[1] || "png";
    const filename = baseFilename + "." + extension;
    
    return processBase64File(filename, mimeType, parts[1]);
  } catch (err) {
    console.error("Base64 candidate upload failed: " + baseFilename, err);
    return "Upload Error: " + err.toString();
  }
}

/**

 * have never been initialized, ensuring zero-config first-run behavior.
 *
 * Routing Trigger: case "GET_FEE_CONFIG" in doPost()
 *
 * Script Properties Keys:
 *   - MONTHLY_FEE: The per-month tuition fee charged to students

function handleGetFeeConfig() {

      monthlyFee: monthlyFee,

    });

      status: "error",

/**

/**
 * BULK_LOG_PAYMENTS Handler (Sharded Transactional Bulk-Writer)
 * --------------------------------------------------
 * Accepts an incoming array of month payment objects, dynamically resolves
 * the target annual shard tab name as "Logs_" + transactionYear, auto-creates
 * the tab with schema headers if it doesn't exist, then injects all rows
 * simultaneously using a single high-speed .setValues() call to bypass
 * timeout gates.
 *
 * Routing Trigger: case "BULK_LOG_PAYMENTS" in doPost()
 *
 * @param {Array<Object>} payloadArray - Array of payment objects, each containing:
 *   {
 *     TXN_ID: string,          // Unique transaction trace token (e.g. "TXN-2026-A3F7B")
 *     TIMESTAMP: string,       // ISO date string of payment time
 *     STUDENT_ID: string,      // Linked student identifier
 *     RL_NO: string,           // Roll number reference
 *     STUDENT_NAME: string,    // Full candidate name
 *     FEE_PERIOD: string,      // Month-Year label (e.g. "January-2026")
 *     STATUS: string,          // Payment status (e.g. "PAID")
 *     AMOUNT_COLLECTED: number // Amount in INR
 *   }
 *
 * Returns: { status: "success", success: true, message: "...", rowsWritten: N }
 */
function handleBulkLogPayments(payloadArray) {
  try {
    // Input Validation Gate
    if (!payloadArray || !Array.isArray(payloadArray) || payloadArray.length === 0) {
      return createResponse({
        status: "error",
        success: false,
        message: "Invalid payload: Expected a non-empty array of payment objects."
      });
    }

    var ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID);
    var schema = SHEET_TEMPLATES.PaymentLog;

    // -------------------------------------------------------
    // PHASE A: Parse the transaction year from FEE_PERIOD.
    // Handles both "2026-06" (New Standard) and "June-2026" (Legacy)
    // -------------------------------------------------------
    var transactionYear = null;
    var firstEntry = payloadArray[0];

    if (firstEntry.FEE_PERIOD && typeof firstEntry.FEE_PERIOD === 'string') {
      var periodParts = firstEntry.FEE_PERIOD.split("-");
      if (periodParts.length >= 2) {
        // First try the new standard (YYYY-MM)
        var parsedYear = parseInt(periodParts[0], 10);
        
        // If it fails, try the legacy standard (Month-YYYY)
        if (isNaN(parsedYear) || parsedYear < 2000) {
            parsedYear = parseInt(periodParts[periodParts.length - 1], 10);
        }

        if (!isNaN(parsedYear) && parsedYear >= 2000 && parsedYear <= 2099) {
          transactionYear = parsedYear;
        }
      }
    }

    // Fallback: derive year from TIMESTAMP
    if (transactionYear === null && firstEntry.TIMESTAMP) {
      var tsDate = new Date(firstEntry.TIMESTAMP);
      if (!isNaN(tsDate.getTime())) {
        transactionYear = tsDate.getFullYear();
      }
    }

    // Ultimate fallback: current server year
    if (transactionYear === null) {
      transactionYear = new Date().getFullYear();
    }

    // -------------------------------------------------------
    // PHASE B: Resolve target shard sheet — Convention over Configuration
    // -------------------------------------------------------
    var sheetName = "Logs_" + transactionYear;
    var sheet = ss.getSheetByName(sheetName);

    // Auto-Creation Guard: If shard tab doesn't exist, create it with schema headers
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      sheet.appendRow(schema);
      sheet.setFrozenRows(1);
      sheet.getRange(1, 1, 1, schema.length).setFontWeight("bold");
    }

    // Auto-Healing: If shard exists but is completely empty, re-inject headers
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(schema);
      sheet.setFrozenRows(1);
      sheet.getRange(1, 1, 1, schema.length).setFontWeight("bold");
    }

    // -------------------------------------------------------
    // PHASE C: Build the 2D matrix aligned to schema column order
    // -------------------------------------------------------
    var matrix = [];
    for (var i = 0; i < payloadArray.length; i++) {
      var entry = payloadArray[i];
      var row = [];
      for (var j = 0; j < schema.length; j++) {
        var key = schema[j];
        if (key === 'TIMESTAMP' && entry[key]) {
          // Ensure timestamps are stored as proper Date objects for sheet formatting
          row.push(new Date(entry[key]));
        } else {
          row.push(entry[key] !== undefined && entry[key] !== null ? entry[key] : "");
        }
      }
      matrix.push(row);
    }

    // -------------------------------------------------------
    // PHASE D: ⚡ Atomic Batch Write — Single execution boundary
    // -------------------------------------------------------
    var lastRow = sheet.getLastRow();
    sheet.getRange(lastRow + 1, 1, matrix.length, schema.length).setValues(matrix);
    
    return createResponse({
      status: "success",
      success: true,
      message: "Successfully logged " + matrix.length + " payment transaction(s) to " + sheetName + ".",
      rowsWritten: matrix.length,
      targetSheet: sheetName
    });
    
  } catch (error) {
    return createResponse({
      status: "error",
      success: false,
      message: "Bulk Payment Write Error: " + error.toString()
    });
  }
}

/**
 * LAZY_FETCH_STUDENT_LOGS Handler (Targeted Single-Candidate Log Fetcher)
 * --------------------------------------------------
 * Fetches payment log records for a single specific student from the annual
 * shard tab matching the targetYear parameter. Uses a precise column-restricted
 * TextFinder search on Column C (STUDENT_ID) with matchEntireCell(true) to
 * extract only that student's payment rows for the given year.
 *
 * This is designed for lazy backward-chaining waterfall lookups — the frontend
 * calls this API once per year, walking backwards through time to build a
 * continuous payment timeline until it hits the admission month.
 *
 * Routing Trigger: case "LAZY_FETCH_STUDENT_LOGS" in doPost()
 *
 * @param {string} studentId - The STUDENT_ID to search for in Column C
 * @param {number|string} targetYear - The year to target (e.g., 2026, "2025")
 * @returns {Object} { status: "success", data: Array<Object>, targetYear: Number, sheetName: String }
 *   Each object in data matches: { TXN_ID, TIMESTAMP, STUDENT_ID, RL_NO, STUDENT_NAME, FEE_PERIOD, STATUS, AMOUNT_COLLECTED }
 */
function handleLazyFetchStudentLogs(studentId, targetYear) {
  try {
    // Input Validation
    if (!studentId || studentId === "") {
      return createResponse({
        status: "error",
        message: "Missing required parameter: studentId."
      });
    }

    var year = parseInt(targetYear, 10);
    if (isNaN(year) || year < 2000 || year > 2099) {
      return createResponse({
        status: "error",
        message: "Invalid targetYear. Must be a 4-digit year between 2000 and 2099."
      });
    }

    var sheetName = "Logs_" + year;
    var schema    = SHEET_TEMPLATES.PaymentLog;

    // ─────────────────────────────────────────────────────────────────────
    // 🚀 ADVANCED SHEETS API v4 — Payment Ledger Historical Read
    // Fetches the entire annual shard in a single REST call instead of
    // opening the Spreadsheet object, probing lastRow, and issuing
    // per-row getRange() reads inside a TextFinder loop.
    //
    // Strategy: Read the full shard once → filter client-side in JS RAM.
    // This eliminates N individual getRange() round-trips (one per matched
    // row) and replaces them with a single REST network call.
    //
    // UI Data Contract: Response keys match SHEET_TEMPLATES.PaymentLog
    // exactly.  TIMESTAMP is serialised to an ISO string.
    // ─────────────────────────────────────────────────────────────────────
    var rawData;
    try {
      var response = Sheets.Spreadsheets.Values.get(
        CONFIG.SPREADSHEET_ID,
        "'" + sheetName + "'"
      );
      rawData = (response && response.values) ? response.values : [];
    } catch (sheetNotFoundErr) {
      // The Advanced API throws when the named sheet does not exist.
      // Treat this as an empty-year result — perfectly valid for waterfall.
      return createResponse({
        status: "success",
        data: [],
        targetYear: year,
        sheetName: sheetName,
        message: "No payment log sheet found for year " + year + ". Sheet '" + sheetName + "' does not exist."
      });
    }

    // Header row only (or completely empty) — return empty dataset
    if (rawData.length <= 1) {
      return createResponse({
        status: "success",
        data: [],
        targetYear: year,
        sheetName: sheetName,
        message: "Payment log sheet for " + year + " exists but contains no data rows."
      });
    }

    var headers = rawData[0];

    // Resolve live STUDENT_ID column index from actual headers
    // (schema position is authoritative, but dynamic resolution adds safety)
    var studentIdColIdx = headers.indexOf("STUDENT_ID");
    if (studentIdColIdx === -1) studentIdColIdx = 2; // Fallback to schema index 2 (Column C)

    // -------------------------------------------------------
    // IN-MEMORY STUDENT_ID FILTER
    // Walk all data rows and collect exact STUDENT_ID matches.
    // Replaces the TextFinder + per-row getRange loop entirely.
    // -------------------------------------------------------
    var results = [];
    var targetIdStr = String(studentId).trim();

    for (var i = 1; i < rawData.length; i++) {
      var row = rawData[i];
      var cellId = (studentIdColIdx < row.length) ? String(row[studentIdColIdx]).trim() : "";
      if (cellId !== targetIdStr) continue;

      // Map row array → schema-keyed object (preserving UI Data Contract)
      var record = {};
      for (var j = 0; j < schema.length; j++) {
        var key       = schema[j];
        var cellValue = (j < row.length) ? row[j] : "";

        // Normalise TIMESTAMP: Advanced API returns date-serial or string;
        // coerce to ISO string to maintain frontend parsing contract.
        if (key === "TIMESTAMP" && cellValue !== "") {
          var ts = new Date(cellValue);
          record[key] = isNaN(ts.getTime()) ? String(cellValue) : ts.toISOString();
        } else {
          record[key] = cellValue;
        }
      }
      results.push(record);
    }

    if (results.length === 0) {
      return createResponse({
        status: "success",
        data: [],
        targetYear: year,
        sheetName: sheetName,
        message: "No payment records found for student '" + studentId + "' in " + sheetName + "."
      });
    }

    return createResponse({
      status: "success",
      data: results,
      targetYear: year,
      sheetName: sheetName,
      message: "Found " + results.length + " payment record(s) for student '" + studentId + "' in " + sheetName + "."
    });
  } catch (error) {
    return createResponse({
      status: "error",
      message: "Lazy Fetch Student Logs Error: " + error.toString()
    });
  }
}

/**
 * FETCH_STATEMENT_RANGE Handler
 * --------------------------------------------------
 * Parametric Range Aggregator Backend
 * Dynamically slices cross-year historical logs.
 */
function handleFetchStatementRange(startDateStr, endDateStr) {
  try {
    var start = new Date(startDateStr);
    var end   = new Date(endDateStr);

    if (isNaN(start.getTime()) || isNaN(end.getTime())) {
      return createResponse({ status: 'error', message: 'Invalid date parameters.' });
    }

    // Ensure end date covers the full day
    end.setHours(23, 59, 59, 999);

    var startYear = start.getFullYear();
    var endYear   = end.getFullYear();

    var mergedLogs = [];
    var schema     = SHEET_TEMPLATES.PaymentLog;

    // ─────────────────────────────────────────────────────────────────────
    // 🚀 ADVANCED SHEETS API v4 — Outstanding Dues / Cross-Year Roster Read
    // Iterates each candidate annual shard within the date range and fetches
    // all rows via Sheets.Spreadsheets.Values.get in a single REST call per
    // shard year, replacing SpreadsheetApp.openById → getSheetByName →
    // getLastRow → getRange().getValues() — a 4-step object-model chain per
    // year — with a single lightweight REST call each.
    //
    // UI Data Contract: mergedLogs contains objects whose keys are the
    // canonical SHEET_TEMPLATES.PaymentLog schema headers, identical to
    // the previous SpreadsheetApp implementation.
    // ─────────────────────────────────────────────────────────────────────
    for (var y = startYear; y <= endYear; y++) {
      var sheetName = 'Logs_' + y;
      var rawData;

      try {
        var response = Sheets.Spreadsheets.Values.get(
          CONFIG.SPREADSHEET_ID,
          "'" + sheetName + "'"
        );
        rawData = (response && response.values) ? response.values : [];
      } catch (sheetNotFoundErr) {
        // Advanced API throws when the named sheet does not exist — skip year
        continue;
      }

      // Skip years with only a header row or no data
      if (rawData.length < 2) continue;

      // Use the live header row to resolve TIMESTAMP column index dynamically
      var liveHeaders     = rawData[0];
      var timestampColIdx = liveHeaders.indexOf("TIMESTAMP");
      if (timestampColIdx === -1) timestampColIdx = 1; // Fallback to schema index 1

      for (var i = 1; i < rawData.length; i++) {
        var row = rawData[i];

        // Normalise and parse the TIMESTAMP cell (Advanced API returns strings)
        var rawTs   = (timestampColIdx < row.length) ? row[timestampColIdx] : "";
        var txnDate = new Date(rawTs);
        if (isNaN(txnDate.getTime())) continue; // Skip rows with unparseable dates

        if (txnDate >= start && txnDate <= end) {
          // Map row array → schema-keyed object (preserving UI Data Contract)
          var logObj = {};
          for (var j = 0; j < schema.length; j++) {
            logObj[schema[j]] = (j < row.length) ? row[j] : "";
          }
          mergedLogs.push(logObj);
        }
      }
    }

    return createResponse({ status: 'success', data: mergedLogs });
  } catch (error) {
    return createResponse({ status: 'error', message: 'Fetch Range Error: ' + error.toString() });
  }
}

/**

 * 📧 MODULE 9: ZERO-STORAGE RUNTIME EMAIL RELAY
 * ==========================================
 */

/**
 * SEND_RUNTIME_EMAIL_ATTACHMENT Handler
 * -------------------------------------------------------------------
 * Decodes a client-transmitted base64 PDF data chunk string directly
 * into a volatile RAM binary blob, and transmits it via native SMTP
 * email relays. Leaves zero files on Google Drive storage rows.
 *
 * Routing Trigger: case "SEND_RUNTIME_EMAIL_ATTACHMENT" in doPost()
 *
 * @param {Object} payload
 *   {
 *     recipientEmail : string,  // Destination SMTP address
 *     subject        : string,  // Email subject line
 *     bodyHtml       : string,  // HTML body content
 *     base64Pdf      : string,  // Base64-encoded PDF (with or without data URI prefix)
 *     fileName       : string   // (Optional) Attachment filename — defaults to "Receipt_Document.pdf"
 *   }
 * @returns {Object} { status: "success"|"error", message: string }
 */
function handleSendRuntimeEmailAttachment(payload) {
  try {
    if (!payload.recipientEmail || !payload.recipientEmail.trim()) {
      return createResponse({ status: "success", message: "Skipped: Recipient email address is empty." });
    }
    if (!payload.base64Pdf) {
      return createResponse({ status: "error", message: "Execution error: Missing Base64 PDF source stream." });
    }

    // Strip potential URL header text prefixes from the data stream base64 array
    var cleanData = payload.base64Pdf.split(',').pop();
    var decodedBytes = Utilities.base64Decode(cleanData);

    // Hold container purely in local volatile server RAM execution limits
    var pdfBlob = Utilities.newBlob(decodedBytes, "application/pdf", payload.fileName || "Receipt_Document.pdf");

    // Broadcast payload directly via Google's internal mail delivery engine
    MailApp.sendEmail({
      to: payload.recipientEmail,
      subject: payload.subject,
      htmlBody: payload.bodyHtml,
      attachments: [pdfBlob]
    });

    return createResponse({
      status: "success",
      message: "Runtime attachment successfully relayed to recipient."
    });

  } catch (error) {
    return createResponse({
      status: "error",
      message: "Apps Script Runtime Mail Gateway Failure: " + error.toString()
    });
  }
}
